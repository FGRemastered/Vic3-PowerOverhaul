	value = 0
	if = {
		limit = { any_scope_building = { is_building_type = building_subsistence_farm trigger_subsistence_building_uses_tools = no } }

			value = b:building_subsistence_farm.level
			multiply = peasant_capacity_subsistence_farm
			multiply = {
				value = 1
				subtract = b:building_subsistence_farm.occupancy
			}
		}
	}
	if = {
		limit = { any_scope_building = { is_building_type = building_subsistence_orchard trigger_subsistence_building_uses_tools = no } }
		add = {
			value = b:building_subsistence_orchard.level

				subtract = b:building_subsistence_orchard.occupancy

		}
	}
	if = {
		limit = { any_scope_building = { is_building_type = building_subsistence_pasture trigger_subsistence_building_uses_tools = no } }
		add = {
			value = b:building_subsistence_pasture.level
			multiply = peasant_capacity_subsistence_pasture
			multiply = {
				value = 1
				subtract = b:building_subsistence_pasture.occupancy
			}
		}
	}
	if = {
		limit = { any_scope_building = { is_building_type = building_subsistence_fishing_village trigger_subsistence_building_uses_tools = no } }
		add = {
			value = b:building_subsistence_fishing_village.level

			multiply = {
				value = 1
				subtract = b:building_subsistence_fishing_village.occupancy
			}
		}
	}
	if = {
		limit = { any_scope_building = { is_building_type = building_subsistence_rice_farm trigger_subsistence_building_uses_tools = no } }
		add = {
			value = b:building_subsistence_rice_farm.level
			multiply = peasant_capacity_rice
			multiply = {
				value = 1
				subtract = b:building_subsistence_rice_farm.occupancy
			}
		}
	}
	if = {
		limit = { any_scope_building = { is_building_type = building_autarky_farm } }	# trigger_subsistence_building_uses_tools = no
		add = {
			value = b:building_autarky_farm.level

		add = {
			value = b:building_autarky_greenhouse.level

			multiply = {
				value = 1
				subtract = b:building_autarky_greenhouse.occupancy
			}
		}
	}
	if = {
		limit = { state_unemployed_population >= 5000 }
		subtract = {
			value		= state_unemployed_population
			divide		= standard_workforce_ratio
			round_to	= 1000
			divide		= 1000
		}
	}
	round = yes
}
