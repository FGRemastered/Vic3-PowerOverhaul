INJECT:law_subjecthood = {
	on_activate = {
		if = {
			limit = {
				any_primary_culture = { THIS ?= cu:british }
				OR = {
					NOT = { any_primary_culture = { THIS ?= cu:anglo_indian } }
					has_modifier = anglo_indian_cultural_acceptance_modifier_negative
					has_modifier = heritage_group_south_asian_cultural_acceptance_modifier_negative
				}
			}
			effect_add_anglo_indian_acceptance = yes
		}
	}
}
