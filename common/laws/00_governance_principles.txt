REPLACE_OR_CREATE:law_monarchy = {
	group = lawgroup_governance_principles
	icon = "gfx/interface/icons/law_icons/monarchy.dds"
	progressiveness = 0
	disallowing_laws = { law_anarchy }
	is_visible = { NOT = { has_law = law_type:law_colonial_administration } }
	can_enact = { trigger_if = { limit = { is_subject = no } trigger_colonial_government = no } }
	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_governance_principles_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}
	modifier = {
		country_legitimacy_headofstate_add					= 20
		country_legitimacy_ideological_incoherence_mult		= 0.10
		country_aristocrats_pol_str_mult					= 0.50	# Increased
		state_aristocrats_investment_pool_efficiency_mult	= 0.05	# Added
		country_authority_add								=  200
	}
	ai_will_do = {
		exists = ruler
		ruler = { has_monarchist_ideology = yes }
	}
	on_activate = {
		if = {
			limit = {
				c:FRA ?= this
				NOT = { has_variable = one_true_french_dynasty }
				has_dlc_feature = voice_of_the_people_content
				game_date > 1836.1.1
			}
			if = {
				limit = { has_variable = fra_divided_monarchists_freeze }
				remove_variable = fra_divided_monarchists_freeze
			}
			add_journal_entry = { type = je_divided_monarchists }
		}
		hidden_effect = {	# Generates French monarch upon enaction.
			if = {
				limit = {
					c:FRA ?= this
					NOT = { has_variable = one_true_french_dynasty }
					has_dlc_feature = voice_of_the_people_content
					game_date > 1836.1.1
					NOR = {
						ruler_is_bonapartist = yes
						ruler_is_orleanist = yes
						ruler_is_legitimist = yes
					}
				}
				fra_handle_republic_monarch_assignment = yes
			}
		}
		if = {
			limit = {
       	 		has_dlc_feature = ip4_content
				c:SPA ?= this
				is_revolutionary = no
				is_secessionist = no
				game_date > 1836.1.1
			}
			add_modifier = modifier_regency
			custom_tooltip = {
				text = activate_provisional_government_tt
				add_journal_entry = { type = je_spain_provisional_government }
			}
		}
		if = {
			limit = { country_has_primary_culture = cu:vietnamese }
			if = {
				limit = { any_scope_state = { state_region = s:STATE_CAMBODIA } }
				random_scope_state = {
					limit = { state_region = s:STATE_CAMBODIA }
					STATE_CAMBODIA_state_name_assign = yes
				}
			}
			if = {
				limit = { any_scope_state = { state_region = s:STATE_LAOS } }
				random_scope_state = {
					limit = { state_region = s:STATE_LAOS }
					STATE_LAOS_state_name_assign = yes
				}
			}
		}
		crown_land_republic_effect = yes
		if = {
			limit = { has_law = law_type:law_neo_absolutism }
			deactivate_law = law_type:law_neo_absolutism
			activate_law = law_type:law_autocracy
		}
	}
	on_deactivate = {
		if = {
			limit = {
				AND = {
					has_law = law_type:law_subjecthood
					NOR = {
						has_law = law_type:law_theocracy
						is_enacting_law = law_type:law_theocracy
					}
				}
			}
			custom_tooltip = because_subjecthood_is_active_tt
			if = {
				limit = {
					any_interest_group = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							NOT = { type ?= law_type:law_subjecthood }
						}
					}
				}
				ordered_interest_group = {
					limit = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							NOT = { type ?= law_type:law_subjecthood }
						}
					}
					order_by = ig_clout
					ordered_preferred_law = {
						limit = {
							is_citizenship_law = yes
							NOT = { type ?= law_type:law_subjecthood }
						}
						min = 0
						max = 1
						check_range_bounds = no
						prev.owner = { activate_law = prev.type }
					}
				}
			}
			else = { activate_law = law_type:law_national_supremacy }
		}
		remove_regency_modifier = yes
	}
	ai_enact_weight_modifier = { # Petitions
		value = 0
		if = { limit = { ai_has_enact_weight_modifier_journal_entries = yes } add = 750 }
	}
	ai_impose_chance = {
		value = 0
		if = {
			limit = {
				has_law = law_type:law_monarchy
				has_strategy = ai_strategy_reactionary_agenda
				power_bloc ?= { has_identity = identity:identity_ideological_union }
				scope:target_country = { NOT = { has_law = law_type:law_theocracy } }	# Is their law different enough from what we want to impose?
			}
			add = base_impose_law_weight	
			if = {
				limit = { 
					scope:target_country = { has_law = law_type:council_republic } 
					NOT = { has_strategy = ai_strategy_egalitarian_agenda }
				}
				multiply = 5
			}
			else = { multiply = 0.5 }	# AI is less likely to mess with governance principles unless dealing with council republics
		}
		if = { limit = { trigger_BIC_british_dictate = yes } multiply = 0 }
	}
}
