REPLACE:law_state_religion = {
	group = lawgroup_church_and_state
	icon = "gfx/interface/icons/law_icons/state_religion.dds"
	progressiveness = -50
	is_visible = { # Turkey should only unlock this as a possibility if it abolishes the Millet System
		NOR = {
			has_law = law_type:law_millet_system
			has_law = law_type:law_people_of_the_book
		}
	}
	can_enact = { NOT = { ROOT.country_definition.religion ?= rel:atheist } }
	on_activate = {
		effect_handle_religion = yes
		recalculate_pop_ig_support = yes
	}
	modifier = {
		country_clergymen_pol_str_mult							= 0.50	# Changed
		country_authority_add									=  200
		country_state_religion_wages_mult						= 0.10
		country_non_state_religion_wages_mult					=-0.10
		country_acceptance_state_religion_add					=   25
		country_acceptance_shared_religious_trait_add			=    0
		country_acceptance_shared_religious_trait_group_add		=  -10
		country_acceptance_no_shared_religious_trait_add		=  -25	
		country_allow_conversion_violent_hostility_bool			= yes
		country_allow_conversion_cultural_erasure_bool			= yes
		country_allow_conversion_open_prejudice_bool			= yes
		country_allow_conversion_second_rate_citizen_bool		= yes
		country_allow_conversion_full_acceptance_bool			= yes
		state_clergymen_investment_pool_efficiency_mult			= 0.05	# New
	}
	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}
	ai_will_do = {
		exists = ruler
		ruler = {
			OR = {
				has_ideology = ideology:ideology_theocrat
				is_misogynist = yes
			}
		}
	}
	ai_enact_weight_modifier = { # Petitions
		value = 0
		if = { limit = { ai_has_enact_weight_modifier_journal_entries = yes } add = 750 }
	}
	ai_impose_chance = {
		value = 0
		if = {
			limit = {
				has_law = law_type:law_state_religion
				power_bloc ?= { has_identity = identity:identity_religious }
				religion = scope:target_country.religion			
			}
			add = base_impose_law_weight	
			if = {	# Very important for religious blocs
				limit = { scope:target_country = { has_law = law_type:law_freedom_of_conscience } }
				multiply = 2 
			}
			else = { multiply = 5 }
		}
		if = { limit = { trigger_BIC_british_dictate = yes } multiply = 0 }
	}
}
