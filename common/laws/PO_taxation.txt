	}

		state_pop_support_movement_modernizer_mult			= 0.25	# Added
	}
	tax_modifier_high = {
		tax_consumption_add	= 0.80	# Increased	- Doubled
		state_pop_support_movement_modernizer_mult			= 0.50	# Added
	}

		state_pop_support_movement_modernizer_mult			= 1.00	# Added
	}
	on_impose = {}
	ai_will_do = {
		NOR = {	# No system with forced privatisation
			has_law_or_variant = law_type:law_laissez_faire
			has_law_or_variant = law_type:law_extraction_economy
		}
		OR = {
			has_technology_researched = conveyors				# Could theoretically nationalise to save themselves 
			has_law_or_variant = law_type:law_command_economy	# Reliable alternative revenue stream
			gdp < 1000000										# Too small to matter - Minting baseline is highest revenue stream
		}
	}	# AI shouldn't change to consumption-based-taxation as it's likely to wreck their economy
	ai_enact_weight_modifier = { # Petitions
		value = 0
		if = { limit = { has_law_or_variant = law_type:law_command_economy }	add = 250 }
		if = { limit = { ai_has_enact_weight_modifier_journal_entries = yes }	add = 750 }
	}
}
