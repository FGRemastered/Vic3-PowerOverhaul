REPLACE:je_opium = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_foreign_affairs
	is_shown_when_inactive = {
		is_player = yes
		any_market = { OWNER = ROOT }
		NOT = { has_variable = opium_je_done }
		any_scope_state = {
			OR = {
				has_potential_resource = bg_opium_plantations
				has_building = building_opium_greenhouse
			}
		}
	}
	possible = {
		any_scope_building = { OR = { is_building_type = building_opium_plantation is_building_type = building_opium_greenhouse } }	# Added Greenhouse
		NOT = { any_primary_culture = { has_cultural_obsession = opium } }
	}
	complete = {
		any_scope_building = {
			OR = { is_building_type = building_opium_plantation is_building_type = building_opium_greenhouse }
			cash_reserves_ratio > 0.5
			weekly_profit > 0
			occupancy >= 0.75
			level >= 10
		}
		any_market = {
			NOT = { OWNER = ROOT }
			# TODO PRCAL-38034 - change to look at country aggregated exports
			mg:opium ?= { market_goods_exports >= 200 }
			OWNER = { any_primary_culture = { NOT = { has_cultural_obsession = opium } } }
		}
	}
	on_complete = {
		trigger_event = { id = opium_events.1 }
		set_variable = { name = opium_je_done }
	}
	transferable = no
	weight = 100
}
