					# NOR = {

						# was_formed_from = PHI
						# THIS ?= c:BIC
						# was_formed_from = BIC
					# }
				}
			}
			trigger_if		= { limit = { $claimant$ ?= { has_variable = senegal_subject_var } }			region = sr:region_senegal }
