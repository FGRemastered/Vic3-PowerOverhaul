trade_center_construction_allowed = {
	trigger_if = {
		limit = {
			OR = {
				OWNER = { has_law = law_type:law_canton_system } 
				scope:investor_country = { has_law = law_type:law_canton_system }
			}
		}
		custom_tooltip = {
			text = canton_system_trade_center_trigger_tt
			state_region = s:STATE_GUANGDONG
			any_scope_building = {
				is_building_type = building_trade_center
				level_after_queued_constructions <= max_canton_system_level
			}
		}
	}
	trigger_else_if = {
		limit = {
			OR = {
				OWNER = { is_player = yes }
				OWNER = { has_law_or_variant = law_type:law_mercantilism }
				scope:investor_country = { has_law_or_variant = law_type:law_mercantilism }
			}
		}
		OR = {
			trigger_capital_or_market_capital = yes
			is_treaty_port = yes
		}
	}
	trigger_else = {
		NOT = { owner = { has_law = law_type:law_isolationism } }
		trigger_if = {
			limit = { owner != scope:investor_country }
			NOT = { scope:investor_country = { has_law = law_type:law_isolationism } }
		}	
	}
}

			limit = { OWNER = { NOT = { has_technology_researched = railways } } }	# Placeholder for land adjacency
			state = { market_access < 1 }
		}
		custom_tooltip = {
			text = insufficient_convoys.tt
			OWNER = {
				modifier:country_convoy_contribution_to_market_owner_add < 1

					num_convoys_available < num_convoys_required
					any_trade_route = { trade_route_needs_convoys_to_grow = yes }	# TODO
				}
			}
		}
		trigger_if = {
			limit = { NOT = { OWNER = market.OWNER } state = { is_incorporated = yes } }
			custom_tooltip = {
				text = market_owner_insufficient_convoys.tt

					OR = {
						num_convoys_available < num_convoys_required
						any_trade_route = { trade_route_needs_convoys_to_grow = yes }	# TODO
					}
				}
			}
		}
		AND = {
			weekly_profit > 0
			market = { mg:merchant_marine = { market_goods_pricier > 0 } }
		}
	}

	building_has_goods_shortage = no
