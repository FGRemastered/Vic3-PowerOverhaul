amendment_upper_strata_tax_cuts_small = {
	parent = law_per_capita_based_taxation
	allowed_laws = {	# Only applicable to laws with dividends taxes.
		law_proportional_taxation
		law_graduated_taxation
	}
	modifier = {
		building_group_bg_manor_houses_tax_mult						=-0.10
		building_group_bg_financial_districts_tax_mult				=-0.10
		building_group_bg_company_headquarter_tax_mult				=-0.10
		building_group_bg_company_regional_headquarter_tax_mult		=-0.10
		# interest_group_ig_industrialists_approval_add				= 1
		# interest_group_ig_landowners_approval_add					= 1
	}
	sponsor_modifier = {
		interest_group_pol_str_mult									= 0.10
		interest_group_approval_add									= 2
	}
	possible = {
		# always = yes
		trigger_if = {
			limit = {
				OR = {
					is_enacting_law = law_type:law_proportional_taxation
					is_enacting_law = law_type:law_graduated_taxation
				}
			}
			NOR = {
				currently_enacting_law = { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_small }
				currently_enacting_law = { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_medium }
				currently_enacting_law = { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_large }
			}
		}
		trigger_else = {
			NOR = {
				active_law:lawgroup_taxation ?= { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_small }
				active_law:lawgroup_taxation ?= { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_medium }
				active_law:lawgroup_taxation ?= { has_amendment = amendment_type:amendment_upper_strata_tax_cuts_large }
			}
		}
	}
	would_sponsor = {
		OR = {
			# law_stance = { law = law_type:law_per_capita_based_taxation value >= neutral }	# Would also include AF and PB
			has_ideology = ideology:ideology_hierarchic			# LO
			has_ideology = ideology:ideology_hierarchic_modern	# LO
			has_ideology = ideology:ideology_plutocratic		# IND
			has_ideology = ideology:ideology_malthusian			# IND
			has_ideology = ideology:ideology_market_liberal		# INT/PB
			has_trait = grifter
		}
		AND = {
			exists = scope:approval
			scope:approval > 0
		}
	}
	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			OR = {
				law_stance = { law = law_type:law_per_capita_based_taxation	value < approve }
				law_stance = { law = law_type:law_hereditary_bureaucrats	value < approve }
			}
		}
	}
	can_repeal = { legitimacy >= legitimacy_to_repeal_amendment }
	amendment_activism_multiplier = 1.0
}
