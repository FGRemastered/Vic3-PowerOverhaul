revolution_pulse_events.26 = {
	type = country_event
	placement = root
	duration = 3
	title = revolution_pulse_events.26.t
	desc = revolution_pulse_events.26.d
	flavor = revolution_pulse_events.26.f
	category = revolution
	event_image = {
		trigger = { trigger_country_is_jewish = yes }
		video = "europenorthamerica_judaism"
	} 
	event_image = {
		trigger = { trigger_country_is_christian = yes }
		video = "southamerica_christianity"
	}
	event_image = {
		trigger = { trigger_country_is_hindu = yes }
		video = "asia_hinduism_sikhism"
	}
	event_image = {
		trigger = { trigger_country_is_confucian_or_shinto = yes }
		video = "asia_confucianism_shinto"
	}
	event_image = {
		trigger = { trigger_country_is_buddhist = yes }
		video = "asia_buddhism"
	}
	event_image = {
		trigger = { trigger_country_is_muslim = yes }
		video = "middleeast_islam"
	}
	event_image = {
		trigger = { trigger_country_is_animist = yes }
		video = "africa_animism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	cooldown = { days = very_long_modifier_time }
	trigger = {
		any_interest_group = {
			ig_counts_as_marginal = no
			is_insurrectionary = yes
			law_stance = {
				law = law_type:law_state_religion
				value > neutral
			}
		}
		NOT = { has_law_or_variant = law_type:law_state_religion }
		any_civil_war = { civil_war_progress_medium = yes }
	}
	cancellation_trigger = {
		NOT = { any_civil_war = { civil_war_progress_medium = yes } }
	}
	immediate = {
		random_civil_war = {
			limit = { civil_war_progress_medium = yes }
			civil_war_save_scopes = yes
		}
		ig:ig_devout ?= { save_scope_as = devout_ig }
	}
	option = {
		name = revolution_pulse_events.26.a
		default_option = yes
		scope:devout_ig = {
			add_modifier = {
				name = modifier_anticlerical_revolution
				days = normal_modifier_time
			}
		}
		add_radicals = {
			value = medium_radicals
			pop_type = clergymen
		}
		scope:ongoing_revolution = { add_civil_war_progress = revolution_retreat_small }
	}
	option = {
		name = revolution_pulse_events.26.b
		scope:ongoing_revolution = {
			add_civil_war_progress = revolution_advance_medium
		}
		scope:devout_ig = {
			add_modifier = {
				name = modifier_clergy_empowered
				days = normal_modifier_time
			}
		}
	}
}
revolution_pulse_events.29 = {
	type = country_event
	placement = root
	duration = 3
	title = revolution_pulse_events.29.t
	desc = revolution_pulse_events.29.d
	flavor = revolution_pulse_events.29.f
	category = revolution
	event_image = {
		trigger = { trigger_country_is_jewish = yes }
		video = "europenorthamerica_judaism"
	} 
	event_image = {
		trigger = { trigger_country_is_christian = yes }
		video = "southamerica_christianity"
	}
	event_image = {
		trigger = { trigger_country_is_hindu = yes }
		video = "asia_hinduism_sikhism"
	}
	event_image = {
		trigger = { trigger_country_is_confucian_or_shinto = yes }
		video = "asia_confucianism_shinto"
	}
	event_image = {
		trigger = { trigger_country_is_buddhist = yes }
		video = "asia_buddhism"
	}
	event_image = {
		trigger = { trigger_country_is_muslim = yes }
		video = "middleeast_islam"
	}
	event_image = {
		trigger = { trigger_country_is_animist = yes }
		video = "africa_animism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	cooldown = { days = very_long_modifier_time }
	trigger = {
		any_interest_group = {
			ig_counts_as_marginal = no
			is_insurrectionary = yes
			law_stance = {
				law = law_type:law_state_religion
				value > neutral
			}
		}
		any_civil_war = {
			OR = {
				civil_war_progress_low = yes
				civil_war_progress_medium = yes
			}
		}
		NOT = { has_law_or_variant = law_type:law_state_religion }
	}
	cancellation_trigger = {
		NOT = {
			any_civil_war = {
				OR = {
					civil_war_progress_low = yes
					civil_war_progress_medium = yes
				}
			}
		}
	}
	immediate = {
		random_civil_war = {
			limit = {
				OR = {
					civil_war_progress_low = yes
					civil_war_progress_medium = yes
				}
			}
			civil_war_save_scopes = yes
		}
		ig:ig_devout ?= { save_scope_as = devout_ig }
		capital = { save_scope_as = capital_scope }
	}
	option = {
		name = revolution_pulse_events.29.a
		default_option = yes
		scope:ongoing_revolution = { add_civil_war_progress = revolution_advance_medium }
		scope:ongoing_revolution_movement = {
			add_modifier = {
				name = modifier_movement_clergy_support
				days = normal_modifier_time
			}
		}
	}
	option = {
		name = revolution_pulse_events.29.b
		scope:ongoing_revolution = { add_civil_war_progress = revolution_retreat_small }
		add_radicals = {
			value = medium_radicals
			interest_group = scope:devout_ig
		}
	}
	option = {
		name = revolution_pulse_events.29.c
		trigger = {
			scope:devout_ig = { is_insurrectionary = no
			}
			scope:ongoing_revolution ?= { civil_war_progress_medium = yes }
		}
		scope:devout_ig = { join_revolution = yes }
	}
}
